> cohens_d <- function(x, group) {
+     x1 <- x[group == "Yes"]
+     x0 <- x[group == "No"]
+     n1 <- length(x1)
+     n0 <- length(x0)
+     pooled_sd <- sqrt(((n1 - 1) * var(x1) + (n0 - 1) * var(x0))/(n1 + 
+         n0 - 2))
+     (mean(x1) - mean(x0))/pooled_sd
+ }
